"""Serve exactly one unchanged HTML fixture and its script for a bounded audit."""
import argparse
import hashlib
import json
from http.server import BaseHTTPRequestHandler, HTTPServer
from pathlib import Path

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--root', type=Path, required=True)
parser.add_argument('--port', type=int, required=True)
args = parser.parse_args()
routes = {'/': ('text/html; charset=utf-8', (args.root/'index.html').read_bytes()),
          '/reader-controls.js': ('text/javascript; charset=utf-8', (args.root/'reader-controls.js').read_bytes())}


class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        entry = routes.get(self.path)
        self.send_response(200 if entry else 404)
        self.send_header('Content-Type', entry[0] if entry else 'text/plain')
        self.send_header('Content-Length', str(len(entry[1]) if entry else 0))
        self.send_header('Cache-Control', 'no-store')
        self.end_headers()
        if entry:self.wfile.write(entry[1])

    def log_message(self, fmt, *values):
        print(json.dumps({'request':self.requestline,'result':fmt % values}),flush=True)


server = HTTPServer(('127.0.0.1',args.port),Handler)
print(json.dumps({'url':f'http://127.0.0.1:{args.port}/','document_routes':['/'],
 'resource_routes':['/reader-controls.js'],'other_routes':'404, empty body; no other authored document',
 'sha256':{path:hashlib.sha256(body).hexdigest() for path,(_,body) in routes.items()}}),flush=True)
server.serve_forever()
