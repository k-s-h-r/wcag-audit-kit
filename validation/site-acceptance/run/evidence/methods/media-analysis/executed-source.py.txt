from pathlib import Path
import json,subprocess,sys,hashlib,shutil
from collections import Counter
import numpy as np
from PIL import Image
ROOT=Path('/workspace/wcag-audit-kit');OUT=Path('/tmp/wcag-acceptance-media-analysis-001');OUT.mkdir(exist_ok=False)
shutil.copy2(__file__,OUT/'executed-source.py.txt');sys.path.insert(0,str(ROOT/'scripts'));from measure import contrast
source=ROOT/'tests/fixtures/audit-site/clip.webm'
subprocess.run(['ffmpeg','-hide_banner','-loglevel','error','-ss','6','-i',str(source),'-frames:v','1',str(OUT/'frame-6.png')],check=True)
a=np.asarray(Image.open(OUT/'frame-6.png').convert('RGB'))
def dominant(region,predicate):
 values=region.reshape(-1,3);selected=values[predicate(values)]
 rgb,count=Counter(map(tuple,selected.tolist())).most_common(1)[0]
 return {'rgb':rgb,'hex':'#'+''.join(f'{x:02x}' for x in rgb),'pixels':count}
blue=dominant(a[135:205,225:415],lambda x:(x[:,0]<20)&(x[:,1]>60)&(x[:,1]<120)&(x[:,2]>150))
green=dominant(a[245:285,265:375],lambda x:(x[:,0]<70)&(x[:,1]>70)&(x[:,1]<150)&(x[:,2]<80))
for item in [blue,green]:item['contrast_with_white']=contrast(item['hex'],'#ffffff')
(OUT/'frame-colors.json').write_text(json.dumps({'note':'Core colors identified from source regions and actual decoded pixels. Not antialias edge values or flash analysis.','video_sha256':hashlib.sha256(source.read_bytes()).hexdigest(),'blue_button':blue,'green_saved_text':green},indent=2)+'\n')
plan=json.loads(Path('/tmp/wcag-acceptance-audit-001/plan.json').read_text())['cases'];case=next(c for c in plan if c['page_url']=='http://127.0.0.1:4330/media' and c['state']=='initial' and c['environment']=='desktop' and c['criterion']=='4.1.3')
request={'case_id':case['case_id'],'url':case['page_url'],'auth_state':'/tmp/wcag-acceptance-bootstrap-002/storage-state.json','locale':'en-US','settle_ms':1400,'observe':[{'selector':'#playback-state'}],
 'actions':[{'type':'reader.press','keys':['Control_L','Home']},{'type':'reader.nextHeading','expect_contains':'Video instructions'}, {'type':'reader.nextButton','expect_contains':'Play video'},
 {'type':'reader.press','keys':['space'],'expect_contains':'Playing'},{'type':'reader.press','keys':['space'],'expect_contains':'Paused'}]}
p=OUT/'media-request.json';p.write_text(json.dumps(request,indent=2)+'\n')
with (OUT/'native-stdout.txt').open('w') as so,(OUT/'native-stderr.txt').open('w') as se:
 r=subprocess.run([sys.executable,str(ROOT/'scripts/screen_reader_native.py'),'--root','/tmp/wcag-at-root','--request',str(p),'--out',str(OUT/'native')],stdout=so,stderr=se)
print(json.dumps({'blue':blue,'green':green,'native_exit':r.returncode}),flush=True)
